#!/usr/bin/env python3
"""Reproduce the supplemental i386 instruction observations in a fresh sandbox."""
import argparse
import hashlib
import json
from pathlib import Path
import random
import shutil
import struct
import subprocess
import sys
import tempfile

p = argparse.ArgumentParser(description=__doc__)
p.add_argument("--repo", required=True, type=Path)
p.add_argument("--mpfr", required=True, help="existing MPFR 4.2.2 shared library")
a = p.parse_args()
e = a.repo / "docs/implementation/evidence/compatibility-php8512-20261104"
s = Path(tempfile.mkdtemp(prefix="phego-instructions-"))
for source, target in [(e / "instructionsfrac.c.txt", "instructionsfrac.c"),
                       (e / "instruction-verify.py.txt", "verify.py"),
                       (a.repo / "tests/semantics/w10-ops/pow.php", "pow.php")]:
    shutil.copyfile(source, s / target)
r = random.Random(876532019)
b = "".join("4000000000000000 " + struct.pack(">d", r.uniform(-.5, .5)).hex()
            + "\n" for _ in range(200000)).encode()
assert hashlib.sha256(b).hexdigest() == "b3dd766b1dae59340f642b8a13bf9ff5932301a4b574a79aeef085da038d2cfe"
(s / "fracpairs.txt").write_bytes(b)
docker = ["docker", "run", "--rm", "--platform", "linux/386", "--network", "none",
          "--read-only", "--memory", "512m", "--pids-limit", "32", "--cap-drop", "ALL",
          "--security-opt", "no-new-privileges", "--mount", f"type=bind,src={s},dst=/w,readonly",
          "--tmpfs", "/tmp:rw,exec,nosuid,size=64m",
          "i386/php@sha256:07e835693e041c2a79cd7a3a7a0fda4b2a8d65c6f93bd89bb8dad5c616f28049"]
raw = subprocess.run(docker + ["sh", "-c", "gcc -O0 /w/instructionsfrac.c -o /tmp/instructions -lm && /tmp/instructions < /w/fracpairs.txt"],
                     check=True, capture_output=True).stdout
(s / "frac-instructions64.txt").write_bytes(raw)
report = subprocess.run([sys.executable, str(s / "verify.py"), str(s), "--mpfr", a.mpfr],
                        check=True, capture_output=True).stdout
d = json.loads(report)
assert d["mpfr_version"] == "4.2.2"
(s / "mpfr-report.json").write_bytes(report)
native = subprocess.run(docker + ["sh", "-c", "php -n /w/pow.php < /w/frac-counter-pairs.txt"],
                        check=True, capture_output=True).stdout
assert native == (s / "frac-counter-expected.txt").read_bytes()
(s / "frac-counter-pow32.txt").write_bytes(native)
old = json.loads((e / "fraccheck.json").read_text())
assert [r[0] for r in d["rows"]] == [r[0] for r in old["mismatches"]]
print(json.dumps({"sandbox": str(s), "mpfr_version": d["mpfr_version"],
                  "cases": d["cases"], "instruction_differences": d["instruction_differences"],
                  "final_rn53_differences": d["final_rn53_differences"],
                  "php_control_cases": len(native.splitlines()), "php_control_differences": 0,
                  "raw_sha256": hashlib.sha256(raw).hexdigest()}, indent=2))
